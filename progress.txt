==================================================

Current extractor:

from urllib.parse import urlparse
import re
import ipaddress

def extract_url_features(url):
    url = str(url).strip()

    if not url:
        raise ValueError("URL cannot be empty")

    if not url.startswith(("http://", "https://")):
        url_for_parse = "https://" + url
    else:
        url_for_parse = url

    parsed = urlparse(url_for_parse)
    hostname = parsed.hostname or ""

    domain = hostname

    if domain.startswith("www."):
        domain = domain[4:]

    domain_length = len(domain)

    try:
        ipaddress.ip_address(hostname)
        is_domain_ip = 1
    except ValueError:
        is_domain_ip = 0

    if "." in hostname:
        tld = hostname.split(".")[-1]
        tld_length = len(tld)
    else:
        tld_length = 0

    hostname_parts = hostname.split(".")

    no_subdomain = (
        len(hostname_parts) - 2
        if len(hostname_parts) >= 3
        else 0
    )

    obfuscated_chars = re.findall(
        r"%[0-9a-fA-F]{2}",
        url
    )

    no_obfuscated_char = len(obfuscated_chars)

    has_obfuscation = (
        1 if no_obfuscated_char > 0 else 0
    )

    obfuscation_ratio = (
        no_obfuscated_char / len(url)
        if len(url) > 0
        else 0
    )

    no_letters = sum(c.isalpha() for c in url)
    no_digits = sum(c.isdigit() for c in url)

    no_equals = url.count("=")
    no_qmark = url.count("?")
    no_ampersand = url.count("&")

    return {
        "URLLength": len(url),
        "DomainLength": domain_length,
        "IsDomainIP": is_domain_ip,
        "TLDLength": tld_length,
        "NoOfSubDomain": no_subdomain,
        "HasObfuscation": has_obfuscation,
        "NoOfObfuscatedChar": no_obfuscated_char,
        "ObfuscationRatio": obfuscation_ratio,
        "NoOfLettersInURL": no_letters,
        "NoOfDegitsInURL": no_digits,
        "NoOfEqualsInURL": no_equals,
        "NoOfQMarkInURL": no_qmark,
        "NoOfAmpersandInURL": no_ampersand,
        "IsHTTPS": 1 if parsed.scheme.lower() == "https" else 0
    }

IMPORTANT:

@app.route("/predict", methods=["POST"])
def predict():

    url = request.form.get("url", "").strip()

    if not url:
        return render_template(
            "index.html",
            error="Please enter a URL."
        )

    try:
        extracted_features = extract_url_features(url)

        features_df = pd.DataFrame([extracted_features])

        features_df = features_df[features]

        prediction = model.predict(features_df)[0]

        probabilities = model.predict_proba(features_df)[0]

        phishing_probability = probabilities[0]
        legitimate_probability = probabilities[1]

        if prediction == 0:
            result = "PHISHING"
        else:
            result = "LEGITIMATE"

        return render_template(
            "index.html",
            url=url,
            result=result,
            phishing_probability=round(phishing_probability * 100, 2),
            legitimate_probability=round(legitimate_probability * 100, 2)
        )

    except Exception as e:
        return render_template(
            "index.html",
            url=url,
            error="Unable to analyze this URL. Please enter a valid URL."
        )

==================================================
44. EXTRACTOR CAVEATS
==================================================

Current extractor uses:

len(url)

for URLLength.

If a URL does not contain:

http://
https://

we internally prepend:

https://

for parsing.

But URLLength is based on the original stripped input URL, not the internally modified URL.

Domain:

hostname

with leading:

www.

removed.

Subdomain calculation is simplistic:

len(hostname_parts) - 2

for three or more hostname parts.

This can be imperfect for public suffixes such as:

co.uk

If this extractor is changed, the model should ideally be retrained because training and serving feature definitions must remain identical.

TRAINING-SERVING CONSISTENCY IS MORE IMPORTANT THAN ARTIFICIALLY REPRODUCING A HIGH BENCHMARK SCORE.
